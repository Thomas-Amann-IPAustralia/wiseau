# OCR for scanned / image-only PDF pages and image uploads. The default engine is
# RapidOCR (detection + recognition on ONNX Runtime) — torch-free, deterministic,
# and more accurate than plain Tesseract on real-world scans (ADR-013). Tesseract
# (MuPDF's built-in; system binary + language data from the Dockerfile) is the
# zero-dependency fallback used automatically when RapidOCR isn't installed. A
# neural handwriting engine (EasyOCR) is opt-in via requirements-ocr.txt +
# WISEAU_OCR_ENGINE=easyocr, so nothing torch-sized ships by default.
rapidocr-onnxruntime==1.4.4
onnxruntime==1.27.0        # inference runtime for RapidOCR; pin for output stability
